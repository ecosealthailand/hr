Google Apps Script Code 
/**
 * Google Apps Script API endpoint for Petty Cash Vouchers.
 * Deploys as a Web App to interact with Google Sheets.
 */

const SHEET_NAME = 'Vouchers';

function doGet(e) {
  const sheet = getOrCreateSheet();
  const data = sheet.getDataRange().getValues();
  const headers = data[0];
  const rows = [];
  
  for (let i = 1; i < data.length; i++) {
    const row = {};
    for (let j = 0; j < headers.length; j++) {
      row[headers[j]] = data[i][j];
    }
    rows.push(row);
  }
  
  return ContentService.createTextOutput(JSON.stringify(rows))
    .setMimeType(ContentService.MimeType.TEXT);
}

function doPost(e) {
  try {
    const payload = JSON.parse(e.postData.contents);
    const sheet = getOrCreateSheet();
    const headers = sheet.getDataRange().getValues()[0];
    
    // Process items details as list
    let itemSummary = '';
    if (payload.items && payload.items.length > 0) {
      itemSummary = payload.items.map(item => `${item.description} (${item.unit}x${item.price})`).join(', ');
    }
    
    const rowData = {
      'Timestamp': new Date().toISOString(),
      'Voucher No': payload.voucherNo || '',
      'Date of Request': payload.date || '',
      'Requestor': payload.requestBy || '',
      'Department': payload.department || '',
      'Expense Type': payload.voucherType || '',
      'Description Summary': itemSummary,
      'Grand Total': payload.grandTotal || 0,
      'VAT Amount': payload.vatAmount || 0,
      'Supplier Details': payload.supplier || '',
      'Reason': payload.reason || '',
      'Status': payload.status || 'Pending',
      'Full JSON Data': JSON.stringify(payload)
    };
    
    // Check if updating an existing record
    const data = sheet.getDataRange().getValues();
    let rowIndex = -1;
    for (let i = 1; i < data.length; i++) {
      if (data[i][1] === payload.voucherNo) { // Column B is Voucher No
        rowIndex = i + 1;
        break;
      }
    }
    
    const rowValues = headers.map(header => rowData[header] !== undefined ? rowData[header] : '');
    
    if (rowIndex > 0) {
      // Update
      sheet.getRange(rowIndex, 1, 1, headers.length).setValues([rowValues]);
    } else {
      // Append
      sheet.appendRow(rowValues);
    }
    
    return ContentService.createTextOutput(JSON.stringify({ success: true, message: 'Record saved successfully!' }))
      .setMimeType(ContentService.MimeType.TEXT);
      
  } catch (error) {
    return ContentService.createTextOutput(JSON.stringify({ success: false, error: error.toString() }))
      .setMimeType(ContentService.MimeType.TEXT);
  }
}

function doOptions(e) {
  return ContentService.createTextOutput("")
    .setHeader('Access-Control-Allow-Origin', '*')
    .setHeader('Access-Control-Allow-Methods', 'GET, POST, OPTIONS')
    .setHeader('Access-Control-Allow-Headers', 'Content-Type');
}

function getOrCreateSheet() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  let sheet = ss.getSheetByName(SHEET_NAME);
  if (!sheet) {
    sheet = ss.insertSheet(SHEET_NAME);
    const headers = [
      'Timestamp',
      'Voucher No',
      'Date of Request',
      'Requestor',
      'Department',
      'Expense Type',
      'Description Summary',
      'Grand Total',
      'VAT Amount',
      'Supplier Details',
      'Reason',
      'Status',
      'Full JSON Data'
    ];
    sheet.appendRow(headers);
    // Format headers
    const range = sheet.getRange(1, 1, 1, headers.length);
    range.setFontWeight('bold');
    range.setBackground('#005ba3');
    range.setFontColor('#ffffff');
    sheet.setFrozenRows(1);
  }
  return sheet;
}
